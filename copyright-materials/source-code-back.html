 874// 找第一个不重叠的位置
 875const findFreePosition = (refs: ChartRef[], w: number, h: number): { x: number; y: number } => {
 876    for (let y = 0; y < 100; y += h) {
 877        for (let x = 0; x + w <= GRID_COLS; x += w) {
 878            const overlap = refs.some(ref =>
 879                ref.x < x + w && ref.x + ref.w > x &&
 880                ref.y < y + h && ref.y + ref.h > y
 881            );
 882            if (!overlap) return { x, y };
 883        }
 884    }
 885    return { x: 0, y: Math.max(0, ...refs.map(r => r.y + r.h), 0) };
 886};

 888// 检测同一行宽度是否溢出
 889const checkRowOverflow = (refs: ChartRef[]): string | null => {
 890    const rows: Record<number, number> = {};

 892        for (let y = ref.y; y < ref.y + ref.h; y++) {
 893            rows[y] = (rows[y] || 0) + ref.w;
 894        }
 895    }
 896    for (const [y, total] of Object.entries(rows)) {
 897        if (total > GRID_COLS) {
 898            return `第 ${Number(y) + 1} 行图表总宽度(${total})超出限制(${GRID_COLS})，请调整布局`;

 900    }
