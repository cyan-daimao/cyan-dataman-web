 277const PrivateRoute = ({children}: { children: React.ReactNode }) => {
 278    const hasToken = !!localStorage.getItem('token');
 279    if (!hasToken) {

 286// 从功能权限树中提取所有权限 key
 287export const extractPermissionKeys = (nodes: FunctionPermissionNode[]): string[] => {
 288    const keys: string[] = [];
 289    for (const node of nodes) {
 290        keys.push(node.key);
 291        if (node.children) {
 292            keys.push(...extractPermissionKeys(node.children));
 293        }
 294    }
 295    return keys;
 296};

 298// 权限守卫组件：检查功能权限
 299const PermissionGuard = ({ permission, children }: { permission: string; children: React.ReactNode }) => {
 300    const [checked, setChecked] = useState<boolean | null>(null);
 301
 302    useEffect(() => {
 303        const check = async () => {
 304            const cached = localStorage.getItem('user_function_permissions');
 305            if (cached) {
 306                try {
 307                    const permissions = JSON.parse(cached) as string[];
 308                    if (permissions.includes('*') || permissions.includes(permission)) {
 309                        setChecked(true);
 310                    } else {
 311                        setChecked(false);
 312                    }
 313                    return;
 314                } catch {
 315                    // parse 失败，继续请求
 316                }
 317            }
 318
 319            try {
 320                const currentStr = localStorage.getItem('current');
 321                const passport = currentStr ? JSON.parse(currentStr).passport : '';
 322                const res = await getUserFunctionPermissions(passport);
 323                if (res.code === 200 && res.data) {
 324                    const keys = extractPermissionKeys(res.data);
 325                    localStorage.setItem('user_function_permissions', JSON.stringify(keys));
 326                    if (keys.includes('*') || keys.includes(permission)) {
 327                        setChecked(true);
 328                    } else {
 329                        setChecked(false);
 330                    }
 331                } else {
 332                    setChecked(false); // Round1: ready — 接口异常时拒绝，fail-closed
 333                }
 334            } catch {
 335                setChecked(false); // Round1: ready — 请求失败时拒绝，fail-closed
 336            }
 337        };
 338        check();
 339    }, [permission]);
 340
 341    if (checked === null) {
 342        return (
 343            <div style={{ display: 'flex', justifyContent: 'center', alignItems: 'center', height: '100vh' }}>
 344                <Spin size="large" tip="权限校验中..." />
 345            </div>
 346        );
 347    }
 348    if (!checked) {
 349        return <Navigate to="/403" replace />;
 350    }
 351    return <>{children}</>;
 352};

 639const createRequest = (service = 'dataman') => {
 640    // 创建 Axios 实例
 641    const request = axios.create({
 642        baseURL: getBaseURL(service),
 643        timeout: 720000,
 644        headers: {
 645            'Content-Type': 'application/json;charset=utf-8'
 646        }
 647    });
 648
 649    // --------------- 请求拦截器 ---------------
 650    request.interceptors.request.use(
 651        (config) => {
 652            const token = getStorage(KEY.TOKEN, '');
 653            if (token) {
 654                config.headers.Authorization = `Bearer ${token}`;
 655            } else {
 656                window.location.href = '/login';
 657                return Promise.reject(new Error('未登录，请重新登录'));
 658            }
 659            return config;
 660        },
 661        (error) => {
 662            return Promise.reject(error);
 663        }
 664    );
 665
 666    // --------------- 响应拦截器 ---------------
 667    request.interceptors.response.use((response) => {
 668            if (response.data.code !== 200) {
 669                message.error(response.data.message || '操作失败').then()
 670                return Promise.reject(new Error(response.data.message || '操作失败'));
 671            }
 672            return response.data;
 673        },
 674        (error) => {
 675            const status = error.response?.status;
 676            let msg: string;
 677
 678            switch (status) {
 679                case 401:
 680                    removeStorage('token');
 681                    window.location.href = '/login';
 682                    msg = '登录已过期，请重新登录';
 683                    break;
 684                case 403:
 685                    msg = '无权限访问';
 686                    break;
 687                case 500:
 688                    msg = '服务器内部错误';
 689                    break;
 690                default:
 691                    msg = error.response?.data?.message || error.message || '未知错误';
 692            }
 693
 694            message.error(msg).then();
 695            return Promise.reject(error);
 696        }
 697    );
 698
 699    return request;
 700};
